cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" NIMBO_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")
if(NOT NIMBO_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "VERSION must contain a version such as 0.1.0")
endif()
project(Nimbo VERSION ${NIMBO_VERSION} LANGUAGES CXX)
if(WIN32)
    enable_language(RC)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Sql Multimedia)
option(NIMBO_MEDIA "Build the local LiveKit media prototype" OFF)
if(NIMBO_MEDIA)
    if(NOT MSVC)
        message(FATAL_ERROR "The pinned LiveKit Windows binary requires MSVC")
    endif()
    set(NIMBO_LIVEKIT_ROOT "" CACHE PATH "Path to the LiveKit SDK bundle")
    add_library(nimbo_livekit SHARED IMPORTED)
    set_target_properties(nimbo_livekit PROPERTIES
        IMPORTED_IMPLIB "${NIMBO_LIVEKIT_ROOT}/lib/livekit.lib"
        IMPORTED_LOCATION "${NIMBO_LIVEKIT_ROOT}/bin/livekit.dll"
        INTERFACE_INCLUDE_DIRECTORIES "${NIMBO_LIVEKIT_ROOT}/include")
    enable_language(C)
    include(cmake/WindowsOpenH264.cmake)
    set(WABT "${CMAKE_CURRENT_SOURCE_DIR}/.tools/media/wabt-1.0.39")
    set(web_root "${CMAKE_CURRENT_SOURCE_DIR}/web")
    if(NOT EXISTS "${web_root}/package.json")
        set(web_root "${CMAKE_CURRENT_SOURCE_DIR}/../web")
    endif()
    set(DSP "${web_root}/node_modules/@sapphi-red/web-noise-suppressor/dist" CACHE PATH "Web client's original DSP modules")
    set(GEN "${CMAKE_CURRENT_BINARY_DIR}/noise-codegen")
    file(MAKE_DIRECTORY "${GEN}")
    foreach(engine gtcrn rnnoise)
        add_custom_command(OUTPUT "${GEN}/${engine}.c" "${GEN}/${engine}.h" COMMAND "${WABT}/bin/wasm2c.exe" "${DSP}/${engine}.wasm" -n "${engine}" -o "${GEN}/${engine}.c" DEPENDS "${DSP}/${engine}.wasm" VERBATIM)
    endforeach()
    set(RT "${WABT}/share/wabt/wasm2c")
    add_library(nimbo_noise STATIC "${GEN}/gtcrn.c" "${GEN}/rnnoise.c" "${RT}/wasm-rt-impl.c" "${RT}/wasm-rt-mem-impl.c" src/noise.cpp)
    set_target_properties(nimbo_noise PROPERTIES AUTOMOC OFF C_STANDARD 99)
    target_include_directories(nimbo_noise PRIVATE "${GEN}" "${RT}" "${WABT}/include" "${CMAKE_CURRENT_SOURCE_DIR}/.tools/media/simde-0.8.2")
    target_compile_definitions(nimbo_noise PRIVATE WASM_RT_USE_MMAP=0 WASM_RT_MEMCHECK_BOUNDS_CHECK=1 WASM_RT_SKIP_SIGNAL_RECOVERY=1)
    add_library(nimbo_media STATIC src/interface_text.cpp src/media_session.cpp src/media_session.h src/voice_events.h src/background.cpp src/background.h src/capture_encoder.cpp)
    target_include_directories(nimbo_media PUBLIC src)
    target_include_directories(nimbo_media PRIVATE ${Qt6Multimedia_PRIVATE_INCLUDE_DIRS})
    target_link_libraries(nimbo_media PUBLIC Qt6::Multimedia Qt6::Gui nimbo_livekit nimbo_noise ole32 mmdevapi PRIVATE nimbo_h264)
    target_compile_options(nimbo_media PRIVATE /W4 /utf-8)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    add_library(nimbo_desktop STATIC desktop/youtube_surface.cpp desktop/share_picker.cpp desktop/api.cpp desktop/motion.cpp desktop/profile.cpp desktop/window.cpp desktop/workspace.cpp desktop/views.cpp desktop/profile_view.cpp desktop/admin.cpp desktop/keys.cpp desktop/chat_media.cpp desktop/message_actions.cpp)
    target_include_directories(nimbo_desktop PUBLIC desktop src)
    target_compile_options(nimbo_desktop PRIVATE /utf-8)
    target_include_directories(nimbo_desktop PRIVATE ${Qt6Multimedia_PRIVATE_INCLUDE_DIRS})
    target_link_libraries(nimbo_desktop PRIVATE dwmapi PUBLIC nimbo_media Qt6::Network Qt6::Widgets bcrypt crypt32 user32)
    set(webview_sdk "${CMAKE_SOURCE_DIR}/.tools/webview2-1.0.3650.58/build/native")
    if(NOT EXISTS "${webview_sdk}/include/WebView2.h")
        message(FATAL_ERROR "Run scripts/bootstrap-webview2.ps1 before building the Windows client")
    endif()
    target_include_directories(nimbo_desktop PRIVATE "${webview_sdk}/include")
    target_link_libraries(nimbo_desktop PRIVATE "${webview_sdk}/x64/WebView2LoaderStatic.lib" shlwapi version)
    configure_file(desktop/player/video.html "${CMAKE_BINARY_DIR}/assets/player/video.html" COPYONLY)
    qt_add_executable(NimboDesktop WIN32 desktop/main.cpp desktop/NimboDesktop.manifest desktop/NimboDesktop.rc)
    set_source_files_properties(desktop/NimboDesktop.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/desktop/nimbo.ico;${CMAKE_CURRENT_SOURCE_DIR}/desktop/version.h")
    target_compile_options(NimboDesktop PRIVATE /utf-8)
    target_link_libraries(NimboDesktop PRIVATE nimbo_desktop)
endif()

add_library(nimbo_core STATIC
    src/local_store.cpp src/local_store.h
    src/audio_probe.cpp src/audio_probe.h
)
target_include_directories(nimbo_core PUBLIC src)
target_link_libraries(nimbo_core PUBLIC Qt6::Core Qt6::Sql Qt6::Multimedia)

qt_add_executable(Nimbo WIN32
    src/main.cpp
    src/main_window.cpp src/main_window.h
)
target_link_libraries(Nimbo PRIVATE nimbo_core Qt6::Widgets)
target_compile_definitions(Nimbo PRIVATE NIMBO_APP_VERSION="${PROJECT_VERSION}")
if(NIMBO_MEDIA)
    target_sources(Nimbo PRIVATE src/media_dialog.cpp src/media_dialog.h)
    target_link_libraries(Nimbo PRIVATE nimbo_media)
    target_compile_definitions(Nimbo PRIVATE NIMBO_MEDIA)
    qt_add_executable(nimbo_media_probe tests/media_probe.cpp)
    target_link_libraries(nimbo_media_probe PRIVATE nimbo_media)
    target_compile_options(nimbo_media_probe PRIVATE /utf-8)
endif()
if(WIN32)
    configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
    target_sources(Nimbo PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
endif()
if(MSVC)
    target_compile_options(nimbo_core PRIVATE /W4 /utf-8)
    target_compile_options(Nimbo PRIVATE /W4 /utf-8)
else()
    target_compile_options(nimbo_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(Nimbo PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    if(NIMBO_MEDIA)
        qt_add_executable(nimbo_desktop_tests tests/native_desktop_tests.cpp)
        target_link_libraries(nimbo_desktop_tests PRIVATE nimbo_desktop Qt6::Test)
        target_compile_options(nimbo_desktop_tests PRIVATE /utf-8)
    endif()
    qt_add_executable(nimbo_tests
        tests/nimbo_tests.cpp
        src/main_window.cpp src/main_window.h
    )
    target_link_libraries(nimbo_tests PRIVATE nimbo_core Qt6::Widgets Qt6::Test)
    if(MSVC)
        target_compile_options(nimbo_tests PRIVATE /utf-8)
    endif()
    add_test(NAME nimbo_tests COMMAND nimbo_tests -o "${CMAKE_CURRENT_BINARY_DIR}/test-results.xml,junitxml")
    set_tests_properties(nimbo_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
